# Additional deps for the agent. google-adk itself is installed by the
# generated Dockerfile at the pinned --adk_version, so do not list it here.
# ADK's --trace_to_cloud / --otel_to_cloud lazy-import these at startup;
# without them the container exits before the port opens. Equivalent to the
# google-adk[otel-gcp] extras.
opentelemetry-exporter-otlp-proto-http
opentelemetry-exporter-gcp-logging
opentelemetry-exporter-gcp-trace
opentelemetry-exporter-gcp-monitoring
opentelemetry-resourcedetector-gcp
opentelemetry-instrumentation-google-genai

# Battery Registry: ground truth is read from Firestore at request time, not
# from a bundled file. Not pulled in by google-adk's default install.
google-cloud-firestore

# Evidence Archive: the payload is written to GCS after the Firestore
# transaction commits, so the chain never advances without a durable object.
google-cloud-storage

# Memory Bank: Vertex AI reasoning-engine working memory. Uses the REST API
# via google.auth.transport.requests.AuthorizedSession — no extra SDK.
requests
